<script>
(function () {
  "use strict";

  /* ====== LA SEULE LIGNE À MODIFIER ======
     Collez ici l'adresse de votre page de capture (étape 2 du tunnel). */
  var PAGE_CAPTURE = "https://VOTRE-COMPTE.systeme.io/capture-diagnostic";
  /* ======================================= */

  var STORE  = "ladiag_state_v1";
  var RESULT = "ladiag_result_v1";

  /* Étape 1 : la situation (le tabac se mesure avec les 2 questions de l'indice HSI) */
  var CONTEXT = [
    { id: "prenom", type: "text", stem: "Pour commencer, quel est votre prénom ?", placeholder: "Votre prénom" },
    { id: "age", type: "one", stem: "Quel âge avez-vous ?",
      options: ["Moins de 30 ans", "30 à 39 ans", "40 à 49 ans", "50 à 59 ans", "60 ans et plus"] },
    { id: "cigarettes", type: "one", stem: "Combien de cigarettes fumez-vous par jour ?",
      options: [
        { t: "10 ou moins", v: 0 },
        { t: "11 à 20", v: 1 },
        { t: "21 à 30", v: 2 },
        { t: "Plus de 30", v: 3 },
        { t: "J'ai arrêté récemment, mais l'envie est toujours là", v: 0 }
      ] },
    { id: "premiere", type: "one", stem: "Combien de temps après votre réveil fumez-vous votre première cigarette ?",
      options: [
        { t: "Dans les 5 minutes", v: 3 },
        { t: "Entre 6 et 30 minutes", v: 2 },
        { t: "Entre 31 et 60 minutes", v: 1 },
        { t: "Après 60 minutes", v: 0 },
        { t: "Je ne fume plus en ce moment", v: 0 }
      ] },
    { id: "anciennete", type: "one", stem: "Depuis combien de temps fumez-vous ?",
      options: ["Moins de 5 ans", "5 à 10 ans", "10 à 20 ans", "Plus de 20 ans"] },
    { id: "tentatives", type: "one", stem: "Avez-vous déjà essayé d'arrêter de fumer ?",
      options: ["Jamais", "Une fois", "2 ou 3 fois", "Plus de 3 fois"] },
    { id: "crainte", type: "many", max: 2,
      stem: "Si vous arrêtiez demain, qu'est-ce qui vous ferait le plus peur ?",
      note: "Deux réponses maximum. On comparera votre ressenti avec vos scores.",
      options: [
        { t: "Ne plus savoir comment gérer mon stress", p: "stress" },
        { t: "Être irritable, nerveuse avec mes proches", p: "stress" },
        { t: "Le manque physique de la cigarette", p: "tabac" },
        { t: "Perdre ma seule vraie pause à moi", p: "tabac" },
        { t: "Prendre du poids", p: "sucre" },
        { t: "Remplacer la cigarette par le sucre", p: "sucre" },
        { t: "Échouer encore une fois", p: "culpabilite" }
      ] }
  ];

  /* Étapes 2 à 5 : les quatre maillons du cercle. 5 = totalement vrai = maillon lourd. */
  var PILLARS = [
    { key: "stress", n: 1, name: "Votre stress",
      intro: "Le point de départ du cercle : ce qui déclenche l'envie.",
      items: [
        "Je me sens souvent débordée par tout ce que j'ai à gérer (travail, famille, charge mentale).",
        "J'ai du mal à me détendre sans cigarette ni grignotage.",
        "Je dors mal ou je me réveille déjà fatiguée."
      ] },
    { key: "tabac", n: 2, name: "La cigarette",
      intro: "La première béquille : ce qu'elle vous apporte, et quand.",
      items: [
        "Quand je suis stressée ou contrariée, la cigarette est mon premier réflexe.",
        "Certains moments me semblent impossibles sans cigarette (le café, la pause, la fin de journée).",
        "La cigarette est souvent la seule vraie pause que je m'accorde."
      ] },
    { key: "sucre", n: 3, name: "Le sucre",
      intro: "La deuxième béquille : celle qui prend le relais quand ça déborde.",
      items: [
        "Quand ça déborde, je me tourne vers le sucré, même sans avoir faim.",
        "Le soir ou après une journée difficile, j'ai du mal à me passer de sucré.",
        "Quand je fume moins, j'ai encore plus envie de sucré."
      ] },
    { key: "culpabilite", n: 4, name: "La culpabilité",
      intro: "Ce qui relance le cercle : le regard que vous portez sur vous.",
      items: [
        "Après une cigarette ou un grignotage, je m'en veux.",
        "Je me dis souvent : « J'ai déjà essayé, je n'y arriverai pas. »",
        "J'ai l'impression d'être seule face à tout ça."
      ] }
  ];

  /* Dernière étape : le projet */
  var PROJECT = [
    { id: "motivation", type: "many", max: 2,
      stem: "Pourquoi voulez-vous changer maintenant ?",
      note: "Deux réponses maximum.",
      options: [
        { t: "Pour ma santé" }, { t: "Pour mes enfants, ma famille" },
        { t: "Pour retrouver du souffle et de l'énergie" }, { t: "Pour être fière de moi" },
        { t: "Pour faire des économies" }, { t: "Pour un projet de grossesse" },
        { t: "Sur le conseil d'un médecin" }
      ] },
    { id: "prete", type: "scale", stem: "À quel point vous sentez-vous prête à être accompagnée pour y arriver ?",
      ends: ["Pas encore", "Je suis prête"] },
    { id: "delai", type: "one", stem: "Quand aimeriez-vous commencer ?",
      options: ["Le plus vite possible", "Dans les prochaines semaines", "Dans quelques mois", "Je me renseigne seulement"] },
    { id: "format", type: "one", stem: "Quel format vous conviendrait le mieux ?",
      options: ["Au cabinet", "En ligne (visio)", "Un mélange des deux", "Je ne sais pas encore"] },
    { id: "sante", type: "one",
      stem: "Une situation de santé particulière vous concerne-t-elle ? (grossesse, problème cardiaque, trouble alimentaire, suivi pour anxiété ou dépression)",
      note: "Cela ne vous empêche pas d'être accompagnée : cela permet d'adapter le suivi, en lien avec votre médecin si besoin.",
      options: ["Non", "Oui", "Je préfère en parler de vive voix"] }
  ];

  var TOTAL_STEPS = 2 + PILLARS.length;
  var answers = {};
  var step = 0;

  var root  = document.getElementById("la-diag");
  var flow  = document.getElementById("laFlow");
  var intro = document.getElementById("laIntro");
  var rail  = document.getElementById("laRail");
  var prog  = document.getElementById("laProgress");
  var lbl   = document.getElementById("laStep");

  function save() {
    try { localStorage.setItem(STORE, JSON.stringify({ a: answers, s: step })); } catch (e) {}
  }
  function load() {
    try {
      var raw = localStorage.getItem(STORE);
      if (!raw) return null;
      var d = JSON.parse(raw);
      if (d && d.a && typeof d.s === "number") return d;
    } catch (e) {}
    return null;
  }
  function esc(s) {
    return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
  }
  function optText(o) { return typeof o === "string" ? o : o.t; }

  function buildRail() {
    rail.innerHTML = "";
    for (var i = 0; i < TOTAL_STEPS; i++) {
      var d = document.createElement("div");
      d.className = "la-seg";
      rail.appendChild(d);
    }
  }
  function paintRail() {
    var segs = rail.children;
    for (var i = 0; i < segs.length; i++) segs[i].className = i < step ? "la-seg la-done" : "la-seg";
    lbl.textContent = "Étape " + Math.min(step, TOTAL_STEPS) + " / " + TOTAL_STEPS;
  }

  function scaleHTML(name, value, ends) {
    var h = '<div class="la-pills">';
    for (var v = 1; v <= 5; v++) {
      var id = "la_" + name + "_v" + v;
      h += '<input type="radio" name="' + name + '" id="' + id + '" value="' + v + '"' +
           (String(value) === String(v) ? " checked" : "") + '><label for="' + id + '">' + v + '</label>';
    }
    ends = ends || ["Pas du tout vrai", "Totalement vrai"];
    return h + '</div><div class="la-ends"><span>' + ends[0] + '</span><span>' + ends[1] + '</span></div>';
  }

  function questionHTML(q) {
    var h = '<div class="la-q">';
    if (q.type === "text") {
      h += '<label class="la-stem" for="la_' + q.id + '">' + q.stem + '</label>' +
           '<input type="text" id="la_' + q.id + '" name="' + q.id + '" autocomplete="given-name" placeholder="' +
           esc(q.placeholder) + '" value="' + esc(answers[q.id] || "") + '">';
    } else if (q.type === "scale") {
      h += '<fieldset><legend class="la-stem">' + q.stem + '</legend>' + scaleHTML(q.id, answers[q.id], q.ends) + '</fieldset>';
    } else if (q.type === "one") {
      h += '<fieldset><legend class="la-stem">' + q.stem + '</legend><div class="la-choices">';
      q.options.forEach(function (o, i) {
        var t = optText(o), id = "la_" + q.id + "_" + i;
        h += '<input type="radio" name="' + q.id + '" id="' + id + '" value="' + esc(t) + '"' +
             (answers[q.id] === t ? " checked" : "") + '><label for="' + id + '">' + esc(t) + '</label>';
      });
      h += '</div>' + (q.note ? '<p class="la-hint" style="margin-top:10px">' + q.note + '</p>' : "") + '</fieldset>';
    } else {
      var picked = answers[q.id] || [];
      h += '<fieldset><legend class="la-stem">' + q.stem + '</legend>' +
           '<div class="la-choices la-multi" data-la-max="' + q.max + '" data-la-group="' + q.id + '">';
      q.options.forEach(function (o, i) {
        var id = "la_" + q.id + "_" + i, on = picked.indexOf(o.t) > -1;
        h += '<input type="checkbox" name="' + q.id + '" id="' + id + '" value="' + esc(o.t) + '"' +
             (on ? " checked" : (picked.length >= q.max ? " disabled" : "")) +
             '><label for="' + id + '">' + esc(o.t) + '</label>';
      });
      h += '</div><p class="la-hint" style="margin-top:10px">' + q.note + '</p></fieldset>';
    }
    return h + '</div>';
  }

  function navHTML(back) {
    return '<div class="la-nav">' +
      '<button class="la-btn" type="button" data-la="next">Continuer</button>' +
      (back ? '<button class="la-btn la-ghost" type="button" data-la="back">Retour</button>' : "") +
      '<span class="la-spacer"></span>' +
      '<p class="la-nag" data-la-nag hidden>Il manque une réponse.</p></div>';
  }

  function renderList(title, lede, list, back) {
    var h = '<section class="la-screen la-anim"><h2 class="la-h2">' + title + '</h2><p class="la-lede">' + lede + '</p>';
    list.forEach(function (q) { h += questionHTML(q); });
    return h + navHTML(back) + '</section>';
  }

  function renderPillar(p) {
    var h = '<section class="la-screen la-anim"><p class="la-kicker">Maillon ' + p.n + ' sur 4</p>' +
            '<h2 class="la-h2">' + p.name + '</h2><p class="la-lede">' + p.intro + '</p>' +
            '<p class="la-hint" style="margin-bottom:26px">Pour chaque phrase : <b>1</b> = pas du tout vrai pour moi, <b>5</b> = totalement vrai.</p>';
    p.items.forEach(function (txt, i) {
      var name = p.key + "_" + i;
      h += '<div class="la-q"><fieldset><legend class="la-stem">' + txt + '</legend>' +
           scaleHTML(name, answers[name]) + '</fieldset></div>';
    });
    return h + navHTML(true) + '</section>';
  }

  function renderGate() {
    var first = answers.prenom ? String(answers.prenom).trim() : "";
    return '<section class="la-screen la-anim">' +
      '<p class="la-kicker">Diagnostic terminé</p>' +
      '<h2 class="la-h2">' + (first ? esc(first) + ', vos résultats sont prêts.' : 'Vos résultats sont prêts.') + '</h2>' +
      '<p class="la-lede">Vos quatre scores sont calculés, avec votre niveau de dépendance à la nicotine, le maillon par lequel commencer et un premier pas concret.</p>' +
      '<p class="la-lede">Indiquez-moi simplement où vous les envoyer : vous les découvrez à l\'écran juste après.</p>' +
      '<div class="la-nav">' +
        '<button class="la-btn" type="button" data-la="go">Voir mes résultats</button>' +
        '<button class="la-btn la-ghost" type="button" data-la="back">Retour</button>' +
      '</div></section>';
  }

  function show(html) {
    intro.hidden = true;
    flow.innerHTML = html;
    prog.hidden = (step < 1 || step > TOTAL_STEPS);
    paintRail();
    var t = flow.querySelector('input[type="text"]');
    if (t) { try { t.focus({ preventScroll: true }); } catch (e) {} }
    root.scrollIntoView({ behavior: "smooth", block: "start" });
  }

  function render() {
    if (step === 0) { flow.innerHTML = ""; intro.hidden = false; prog.hidden = true; return; }
    if (step === 1) return show(renderList("Votre situation aujourd'hui", "Quelques questions rapides pour faire connaissance.", CONTEXT, false));
    if (step >= 2 && step <= 1 + PILLARS.length) return show(renderPillar(PILLARS[step - 2]));
    if (step === TOTAL_STEPS) return show(renderList("Votre projet", "Dernière étape : ce que vous voulez, et comment.", PROJECT, true));
    show(renderGate());
  }

  function collect() {
    var t = flow.querySelector('input[type="text"]');
    if (t) answers[t.name] = t.value;
    var radios = flow.querySelectorAll('input[type="radio"]:checked');
    for (var i = 0; i < radios.length; i++) answers[radios[i].name] = radios[i].value;
    var groups = flow.querySelectorAll(".la-multi");
    for (var g = 0; g < groups.length; g++) {
      var name = groups[g].getAttribute("data-la-group"), picked = [];
      var boxes = groups[g].querySelectorAll('input[type="checkbox"]:checked');
      for (var b = 0; b < boxes.length; b++) picked.push(boxes[b].value);
      answers[name] = picked;
    }
  }

  function missingIn(list) {
    for (var i = 0; i < list.length; i++) {
      var q = list[i];
      if (q.type === "text") {
        var t = flow.querySelector('input[name="' + q.id + '"]');
        if (t && !t.value.trim()) return t;
      } else if (!flow.querySelector('input[name="' + q.id + '"]:checked')) {
        return flow.querySelector('input[name="' + q.id + '"]');
      }
    }
    return null;
  }

  function missing() {
    if (step === 1) return missingIn(CONTEXT);
    if (step === TOTAL_STEPS) return missingIn(PROJECT);
    if (step >= 2 && step <= 1 + PILLARS.length) {
      var p = PILLARS[step - 2];
      for (var k = 0; k < 3; k++) {
        var n = p.key + "_" + k;
        if (!flow.querySelector('input[name="' + n + '"]:checked')) return flow.querySelector('input[name="' + n + '"]');
      }
    }
    return null;
  }

  function pointsOf(qid) {
    var val = answers[qid], pts = 0;
    CONTEXT.forEach(function (q) {
      if (q.id !== qid) return;
      q.options.forEach(function (o) { if (o.t === val) pts = o.v; });
    });
    return pts;
  }

  function buildResult() {
    var scores = {}, top = null;
    PILLARS.forEach(function (p) {
      var s = 0;
      for (var i = 0; i < 3; i++) s += Number(answers[p.key + "_" + i] || 0);
      var pct = Math.round((s - 3) / 12 * 100);
      scores[p.key] = pct;
      if (!top || pct > top.pct) top = { k: p.key, pct: pct, name: p.name };
    });
    var hsi = pointsOf("cigarettes") + pointsOf("premiere");
    var dep = hsi <= 1 ? "faible" : (hsi <= 4 ? "moyenne" : "forte");
    var plusieurs = answers.tentatives === "2 ou 3 fois" || answers.tentatives === "Plus de 3 fois";

    var profil;
    if (scores.tabac >= 50 && scores.sucre >= 50) profil = "duo";
    else if (plusieurs && scores.culpabilite >= 50) profil = "cercle";
    else profil = "soupape";

    var craintes = [];
    (answers.crainte || []).forEach(function (txt) {
      CONTEXT.forEach(function (q) {
        if (q.id !== "crainte") return;
        q.options.forEach(function (o) { if (o.t === txt && craintes.indexOf(o.p) === -1) craintes.push(o.p); });
      });
    });

    return {
      prenom: (answers.prenom || "").trim(),
      age: answers.age || "",
      cigarettes: answers.cigarettes || "",
      premiere: answers.premiere || "",
      anciennete: answers.anciennete || "",
      tentatives: answers.tentatives || "",
      craintes: craintes,
      craintesTextes: answers.crainte || [],
      motivation: answers.motivation || [],
      prete: Number(answers.prete || 0),
      delai: answers.delai || "",
      format: answers.format || "",
      sante: answers.sante || "",
      scores: scores,
      hsi: hsi,
      dependance: dep,
      top: top.k,
      topName: top.name,
      profil: profil,
      ts: Date.now()
    };
  }

  function goNext() {
    collect();
    var miss = missing();
    var nag = flow.querySelector("[data-la-nag]");
    if (miss) {
      if (nag) nag.hidden = false;
      var box = miss.closest(".la-q") || miss;
      box.scrollIntoView({ behavior: "smooth", block: "center" });
      if (miss.type === "text") miss.focus();
      return;
    }
    if (nag) nag.hidden = true;
    step += 1;
    if (step === TOTAL_STEPS + 1) {
      try { localStorage.setItem(RESULT, JSON.stringify(buildResult())); } catch (e) {}
    }
    save();
    render();
  }

  root.addEventListener("click", function (e) {
    if (!e.target || !e.target.closest) return;
    var btn = e.target.closest("[data-la]");
    if (btn) {
      var act = btn.getAttribute("data-la");
      if (act === "next") return goNext();
      if (act === "back") { collect(); step = Math.max(0, step - 1); save(); return render(); }
      if (act === "go") { window.location.href = PAGE_CAPTURE; return; }
    }
    if (e.target.id === "laStart") { answers = {}; step = 1; save(); render(); }
    if (e.target.id === "laResume") { var d = load(); if (d) { answers = d.a; step = d.s; render(); } }
  });

  root.addEventListener("change", function (e) {
    var group = e.target.closest ? e.target.closest(".la-multi") : null;
    if (group) {
      var max = Number(group.getAttribute("data-la-max")) || 99;
      var boxes = group.querySelectorAll('input[type="checkbox"]');
      var n = group.querySelectorAll('input[type="checkbox"]:checked').length;
      for (var i = 0; i < boxes.length; i++) boxes[i].disabled = (!boxes[i].checked && n >= max);
    }
    var nag = flow.querySelector("[data-la-nag]");
    if (nag) nag.hidden = true;
  });

  buildRail();
  var saved = load();
  if (saved && saved.s > 0) document.getElementById("laResume").hidden = false;
  render();
})();
</script>
